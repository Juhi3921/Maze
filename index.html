const stepThroughCell = (row, column) => { // If i have visted the cell at [row, column], then return if (grid[row][column]) { return; } // Mark this cell as being visited grid[row][column] = true; // Assemble randomly-ordered list of neighbors const neighbors = shuffle([ [row - 1, column, "up"], [row, column + 1, "right"], [row + 1, column, "down"], [row, column - 1, "left"], ]); // For each neighbor.... for (let neighbor of neighbors) { const [nextRow, nextColumn, direction] = neighbor; // See if that neighbor is out of bounds (kita tidak bisa jalan melewati border) if ( nextRow < 0 || nextRow >= cellsVertical || nextColumn < 0 || nextColumn >= cellsHorizontal ) { continue; } // If we have visited that neighbor, continue to next neighbor if (grid[nextRow][nextColumn]) { continue; } // Remove a wall from either horizontals or verticals if (direction === "left") { verticals[row][column - 1] = true;
